```python
#!/usr/bin/env python3
from __future__ import annotations

import argparse
import hashlib
import json
import re
import shutil
import sys
from datetime import datetime, timezone
from pathlib import Path
from typing import Any


TOOL_VERSION = "0.1.0"
SOURCE_ARCHIVE_SUFFIXES = {".zip", ".tar", ".gz", ".tgz", ".bz2", ".xz", ".7z"}


class CleanroomError(Exception):
    pass


def now_utc() -> str:
    return datetime.now(timezone.utc).isoformat().replace("+00:00", "Z")


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for chunk in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def read_text_manifest(path: Path) -> str:
    raw = path.read_bytes()
    if b"\x00" in raw:
        raise CleanroomError(f"binary or non-text manifest rejected: {path}")
    try:
        return raw.decode("utf-8")
    except UnicodeDecodeError as exc:
        raise CleanroomError(f"binary or non-text manifest rejected: {path}") from exc


def safe_package_path(name: str) -> str:
    safe = re.sub(r"[^A-Za-z0-9._-]+", "-", name.strip())
    safe = safe.strip(".-")
    return safe or "package"


def write_json(path: Path, payload: dict[str, Any] | list[Any]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(payload, indent=2, sort_keys=True) + "\n", encoding="utf-8")


def append_ledger(run_dir: Path, event: str, **fields: Any) -> None:
    payload = {
        "timestamp": now_utc(),
        "event": event,
        "tool_version": TOOL_VERSION,
        **fields,
    }
    with (run_dir / "ledger.jsonl").open("a", encoding="utf-8") as handle:
        handle.write(json.dumps(payload, sort_keys=True) + "\n")


def detect_ecosystem(path: Path) -> str:
    name = path.name.lower()
    if name == "package.json":
        return "npm"
    if name == "requirements.txt":
        return "python"
    if name == "cargo.toml":
        return "cargo"
    raise CleanroomError(f"unsupported manifest type: {path.name}")


def package_record(ecosystem: str, name: str, requested_version: str, dependency_class: str, extras: list[str] | None = None) -> dict[str, Any]:
    return {
        "ecosystem": ecosystem,
        "name": name,
        "requested_version": requested_version,
        "dependency_class": dependency_class,
        "extras": extras or [],
        "source_exposure": "manifest-only",
    }


def parse_package_json(text: str) -> list[dict[str, Any]]:
    try:
        data = json.loads(text)
    except json.JSONDecodeError as exc:
        raise CleanroomError(f"malformed package.json: {exc}") from exc
    records: list[dict[str, Any]] = []
    for dependency_class in ("dependencies", "devDependencies", "peerDependencies", "optionalDependencies"):
        dependencies = data.get(dependency_class, {})
        if not isinstance(dependencies, dict):
            raise CleanroomError(f"package.json field {dependency_class} must be an object")
        for name, version in dependencies.items():
            records.append(package_record("npm", str(name), str(version), dependency_class))
    return records


REQ_RE = re.compile(r"^\s*([A-Za-z0-9_.-]+)(?:\[([A-Za-z0-9_,.-]+)\])?\s*([=!<>~]=?|===)?\s*(.*)$")


def parse_requirements(text: str) -> list[dict[str, Any]]:
    records: list[dict[str, Any]] = []
    for line_number, original_line in enumerate(text.splitlines(), start=1):
        line = original_line.strip()
        if not line or line.startswith("#") or line.startswith("-"):
            continue
        line = line.split(" #", 1)[0].strip()
        match = REQ_RE.match(line)
        if not match:
            raise CleanroomError(f"malformed requirements.txt line {line_number}: {original_line}")
        name, extras_raw, operator, version = match.groups()
        extras = [item.strip() for item in extras_raw.split(",")] if extras_raw else []
        requested_version = f"{operator}{version.strip()}" if operator else ""
        records.append(package_record("python", name, requested_version, "dependencies", extras))
    return records


def parse_cargo_value(raw_value: str) -> str:
    value = raw_value.strip().rstrip(",")
    if value.startswith('"') and '"' in value[1:]:
        return value.split('"', 2)[1]
    version_match = re.search(r'version\s*=\s*"([^"]+)"', value)
    if version_match:
        return version_match.group(1)
    return value


def parse_cargo_toml(text: str) -> list[dict[str, Any]]:
    records: list[dict[str, Any]] = []
    current_section = ""
    allowed_sections = {"dependencies", "dev-dependencies", "build-dependencies"}
    for original_line in text.splitlines():
        line = original_line.strip()
        if not line or line.startswith("#"):
            continue
        if line.startswith("[") and line.endswith("]"):
            current_section = line.strip("[]")
            continue
        if current_section in allowed_sections and "=" in line:
            name, raw_value = line.split("=", 1)
            records.append(package_record("cargo", name.strip(), parse_cargo_value(raw_value), current_section))
    return records


def parse_manifest(path: Path, ecosystem: str) -> list[dict[str, Any]]:
    if path.suffix.lower() in SOURCE_ARCHIVE_SUFFIXES:
        raise CleanroomError(f"source archive ingestion blocked by default: {path.name}")
    text = read_text_manifest(path)
    if ecosystem == "npm":
        return parse_package_json(text)
    if ecosystem == "python":
        return parse_requirements(text)
    if ecosystem == "cargo":
        return parse_cargo_toml(text)
    raise CleanroomError(f"unsupported ecosystem: {ecosystem}")


def load_dependencies(run_dir: Path) -> list[dict[str, Any]]:
    path = run_dir / "dependencies.json"
    if not path.exists():
        raise CleanroomError(f"missing dependencies.json in {run_dir}")
    payload = json.loads(path.read_text(encoding="utf-8"))
    return list(payload.get("dependencies", []))


def load_metadata_fixture(path: Path | None) -> dict[str, Any]:
    if path is None:
        return {}
    return json.loads(path.read_text(encoding="utf-8"))


def command_intake(args: argparse.Namespace) -> int:
    manifest = Path(args.manifest).resolve()
    run_dir = Path(args.out).resolve()
    if not manifest.exists():
        raise CleanroomError(f"manifest not found: {manifest}")
    if run_dir.exists() and any(run_dir.iterdir()):
        raise CleanroomError(f"run directory already exists and is not empty: {run_dir}")
    run_dir.mkdir(parents=True, exist_ok=True)
    append_ledger(run_dir, "manifest.intake.started", manifest=str(manifest))
    ecosystem = detect_ecosystem(manifest)
    dependencies = parse_manifest(manifest, ecosystem)
    shutil.copyfile(manifest, run_dir / "manifest.original")
    manifest_hash = sha256_file(manifest)
    normalized = {
        "schema_version": "1",
        "ecosystem": ecosystem,
        "manifest_path": str(manifest),
        "manifest_sha256": manifest_hash,
        "dependency_count": len(dependencies),
        "tool_version": TOOL_VERSION,
    }
    write_json(run_dir / "manifest.normalized.json", normalized)
    write_json(run_dir / "dependencies.json", {"schema_version": "1", "dependencies": dependencies})
    append_ledger(run_dir, "manifest.intake.completed", ecosystem=ecosystem, dependency_count=len(dependencies), manifest_sha256=manifest_hash)
    print(f"Intake complete: {len(dependencies)} dependencies -> {run_dir}")
    return 0


def command_analyze(args: argparse.Namespace) -> int:
    run_dir = Path(args.run_dir).resolve()
    dependencies = load_dependencies(run_dir)
    fixture = load_metadata_fixture(Path(args.metadata_fixture).resolve() if args.metadata_fixture else None)
    append_ledger(run_dir, "metadata.analysis.started", dependency_count=len(dependencies), live_requested=bool(args.live))
    packages: list[dict[str, Any]] = []
    for dependency in dependencies:
        name = dependency["name"]
        fixture_record = dict(fixture.get(name, {}))
        warnings: list[str] = []
        if args.live:
            warnings.append("live registry resolution was requested but this MVP run uses fixture/offline metadata only")
        if not fixture_record:
            warnings.append("metadata unavailable in offline fixture mode")
        package = {
            "name": name,
            "ecosystem": dependency["ecosystem"],
            "requested_version": dependency["requested_version"],
            "dependency_class": dependency["dependency_class"],
            "license": fixture_record.get("license", "unknown"),
            "homepage": fixture_record.get("homepage"),
            "repository": fixture_record.get("repository"),
            "docs_url": fixture_record.get("docs_url"),
            "latest_version": fixture_record.get("latest_version"),
            "declared_interfaces": fixture_record.get("declared_interfaces", []),
            "metadata_source": "fixture" if fixture_record else "offline-default",
            "source_exposure": "no-source-archive-ingested",
            "warnings": warnings,
        }
        packages.append(package)
        append_ledger(run_dir, "metadata.package.recorded", package=name, metadata_source=package["metadata_source"], warnings=warnings)
    write_json(run_dir / "package-metadata.json", {"schema_version": "1", "packages": packages})
    append_ledger(run_dir, "metadata.analysis.completed", package_count=len(packages))
    print(f"Analysis complete: {len(packages)} package metadata records -> {run_dir / 'package-metadata.json'}")
    return 0


def load_metadata(run_dir: Path) -> dict[str, dict[str, Any]]:
    path = run_dir / "package-metadata.json"
    if not path.exists():
        raise CleanroomError(f"missing package-metadata.json in {run_dir}")
    payload = json.loads(path.read_text(encoding="utf-8"))
    return {item["name"]: item for item in payload.get("packages", [])}


def packet_targets(dependencies: list[dict[str, Any]], requested: str | None) -> list[dict[str, Any]]:
    if requested is None:
        return dependencies
    matches = [item for item in dependencies if item["name"] == requested]
    if not matches:
        raise CleanroomError(f"package not found in dependencies: {requested}")
    return matches


def write_analysis_packet(run_dir: Path, dependency: dict[str, Any], metadata: dict[str, Any]) -> None:
    package_dir = run_dir / "analysis-packets" / safe_package_path(dependency["name"])
    package_dir.mkdir(parents=True, exist_ok=True)
    interfaces = metadata.get("declared_interfaces") or []
    docs_url = metadata.get("docs_url")
    evidence = []
    if docs_url:
        evidence.append({"category": "docs", "ref": docs_url})
    if metadata.get("homepage"):
        evidence.append({"category": "metadata", "ref": metadata["homepage"]})
    analysis = [
        f"# Analysis Packet: {dependency['name']}",
        "",
        "## Clean-Room Boundary",
        "This packet is derived from manifest data and public metadata/docs fixtures recorded in the evidence ledger. Original package source archives were not ingested by this tool.",
        "",
        "## Package Purpose",
        f"- Ecosystem: `{dependency['ecosystem']}`",
        f"- Requested version: `{dependency['requested_version']}`",
        f"- Observed license metadata: `{metadata.get('license', 'unknown')}`",
        "",
        "## Public Interface",
    ]
    if interfaces:
        analysis.extend(f"- `{interface}`" for interface in interfaces)
    else:
        analysis.append("- No declared public interface metadata was available in this run.")
    analysis.extend(
        [
            "",
            "## Behavioral Requirements",
            "- Preserve the public call shapes listed above.",
            "- Create tests from documented examples before writing replacement implementation code.",
            "- Treat undocumented behavior as unknown until a public source records it.",
            "",
            "## Source Evidence",
        ]
    )
    if evidence:
        analysis.extend(f"- {item['category']}: {item['ref']}" for item in evidence)
    else:
        analysis.append("- No public docs URL was available in fixture/offline metadata.")
    (package_dir / "analysis.md").write_text("\n".join(analysis) + "\n", encoding="utf-8")
    write_json(
        package_dir / "public-interface.json",
        {
            "schema_version": "1",
            "package": dependency["name"],
            "interfaces": interfaces,
            "source_categories": ["metadata", "docs"] if docs_url else ["metadata"],
        },
    )
    with (package_dir / "evidence-ledger.jsonl").open("w", encoding="utf-8") as handle:
        for item in evidence:
            handle.write(json.dumps({"timestamp": now_utc(), **item}, sort_keys=True) + "\n")


def write_builder_packet(run_dir: Path, dependency: dict[str, Any], metadata: dict[str, Any]) -> None:
    package_dir = run_dir / "builder-packets" / safe_package_path(dependency["name"])
    package_dir.mkdir(parents=True, exist_ok=True)
    interfaces = metadata.get("declared_interfaces") or []
    readme = [
        f"# Builder Packet: {dependency['name']}",
        "",
        "## Boundary Statement",
        "Implement from this packet and the referenced public documentation only. Do not read original package source archives while using this packet.",
        "",
        "## Required Public Interfaces",
    ]
    if interfaces:
        readme.extend(f"- `{interface}`" for interface in interfaces)
    else:
        readme.append("- No explicit interface metadata was available; start by defining tests for documented behavior.")
    readme.extend(
        [

            "## Implementation Tasks",

            "2. Implement the smallest compatible behavior supported by public docs.",
            "3. Record every public evidence source used during implementation.",
            "4. Run tests before claiming compatibility.",
        ]
    )
    (package_dir / "README.md").write_text("\n".join(readme) + "\n", encoding="utf-8")
    write_json(
        package_dir / "spec.json",
        {
            "schema_version": "1",
            "package": dependency["name"],
            "ecosystem": dependency["ecosystem"],
            "requested_version": dependency["requested_version"],
            "interfaces": interfaces,
            "clean_room_boundary": "public-docs-and-metadata-only",
            "source_archive_ingested": False,
        },
    )
    test_plan = [
        f"# Test Plan: {dependency['name']}",
        "",
        "- Create unit tests for every required public interface.",
        "- Include error and edge cases mentioned by public docs.",
        "- Run tests before comparing behavior claims.",
        "- Record uncovered behavior as unknown instead of guessing.",

def command_packet(args: argparse.Namespace) -> int:
    run_dir = Path(args.run_dir).resolve()
    dependencies = load_dependencies(run_dir)
    metadata_by_name = load_metadata(run_dir)
    targets = packet_targets(dependencies, args.package)
    append_ledger(run_dir, "packet.generation.started", package_count=len(targets))
    for dependency in targets:
        metadata = metadata_by_name.get(dependency["name"], {})
        write_analysis_packet(run_dir, dependency, metadata)
        write_builder_packet(run_dir, dependency, metadata)
        append_ledger(run_dir, "packet.package.generated", package=dependency["name"])
    append_ledger(run_dir, "packet.generation.completed", package_count=len(targets))
    print(f"Packet generation complete: {len(targets)} packages -> {run_dir}")
    return 0


def command_report(args: argparse.Namespace) -> int:
    run_dir = Path(args.run_dir).resolve()
    dependencies = load_dependencies(run_dir)
    metadata_by_name = load_metadata(run_dir) if (run_dir / "package-metadata.json").exists() else {}
    packet_count = len(list((run_dir / "builder-packets").glob("*"))) if (run_dir / "builder-packets").exists() else 0
    append_ledger(run_dir, "report.generation.started", dependency_count=len(dependencies), packet_count=packet_count)
    lines = [
        "# Malus Clean-Room Report",
        "",
        f"- Run directory: `{run_dir}`",
        f"- Dependency count: `{len(dependencies)}`",
        f"- Builder packet count: `{packet_count}`",
        "- Boundary: source archives were not ingested by this tool.",
        "- Claim posture: this report records evidence and packet status; it does not assert legal clearance.",
        "",
        "## Packages",
        "",
        "| Package | Ecosystem | Requested | License Metadata | Source Exposure | Warnings |",
        "| --- | --- | --- | --- | --- | --- |",
    ]
    for dependency in dependencies:
        metadata = metadata_by_name.get(dependency["name"], {})
        warnings = "; ".join(metadata.get("warnings", []))
        lines.append(
            "| {name} | {ecosystem} | {requested} | {license} | {source} | {warnings} |".format(
                name=dependency["name"],
                ecosystem=dependency["ecosystem"],
                requested=dependency["requested_version"] or "unspecified",
                license=metadata.get("license", "unknown"),
                source=metadata.get("source_exposure", dependency.get("source_exposure", "manifest-only")),
                warnings=warnings or "none",
            )
        )
    lines.extend(
        [
            "",
            "## Next Actions",
            "",
            "1. Review analysis packets for evidence quality.",
            "2. Hand builder packets to an implementation worker that has not read original package source archives.",
            "3. Require tests before making compatibility claims.",
        ]
    )
    report_path = run_dir / "MALUS_CLEANROOM_REPORT.md"
    report_path.write_text("\n".join(lines) + "\n", encoding="utf-8")
    append_ledger(run_dir, "report.generation.completed", report=str(report_path))
    print(f"Report written: {report_path}")
    return 0


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(prog="malus_cleanroom", description="Clean-room packet generator for dependency manifests")
    subcommands = parser.add_subparsers(dest="command", required=True)

    intake = subcommands.add_parser("intake", help="Parse a manifest and write dependency records")
    intake.add_argument("manifest")
    intake.add_argument("--out", required=True)
    intake.set_defaults(func=command_intake)

    analyze = subcommands.add_parser("analyze", help="Generate package metadata records")
    analyze.add_argument("run_dir")
    analyze.add_argument("--metadata-fixture")
    analyze.add_argument("--live", action="store_true")
    analyze.set_defaults(func=command_analyze)

    packet = subcommands.add_parser("packet", help="Generate analysis and builder packets")
    packet.add_argument("run_dir")
    packet.add_argument("--package")
    packet.set_defaults(func=command_packet)

    report = subcommands.add_parser("report", help="Write the Markdown run report")
    report.add_argument("run_dir")
    report.set_defaults(func=command_report)

    return parser


def main(argv: list[str] | None = None) -> int:
    parser = build_parser()
    args = parser.parse_args(argv)
    try:
        return int(args.func(args))
    except CleanroomError as exc:
        print(str(exc), file=sys.stderr)
        return 2


if __name__ == "__main__":
    raise SystemExit(main())
```
